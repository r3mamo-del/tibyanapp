// ============================================================================ // طبقة وسيطة (Cloudflare Worker) — تخبّي مفتاح GitHub الحقيقي تمامًا عن أي // حد يشوف كود تطبيق "تبيان" بالمتصفح. التطبيق يتكلم مع هذا الرابط فقط، // وهو يتكلم مع GitHub بالمفتاح السري (مخزَّن كـ Secret بإعدادات Cloudflare // نفسها، غير مرئي إطلاقًا بأي كود أو سجلّ). // // كيف يشتغل (قراءة بيانات المستشفيات): // GET https://<اسم-الـWorker>.workers.dev/data/audit/breast.json // → يجيب نفس الملف من مستودع raqeeb-data الخاص، بدون ما يكشف المفتاح // لأي طرف خارجي إطلاقًا. // // كيف يشتغل (إعدادات مشتركة بكل الأجهزة — كلمة مرور المشرف، المستهدفات): // GET /storage/{key} → يرجّع القيمة المخزَّنة (أو 404 لو ما فيه) // POST /storage/{key} {"value": "..."} → يحفظ/يحدّث القيمة // يعتمد على Cloudflare KV (تخزين مفتاح-قيمة مجاني) — يخلي كل جهاز يفتح // التطبيق يشوف نفس الإعدادات فورًا، بدل تخزين منفصل بكل متصفح على حدة. // // حماية إضافية: يقبل طلبات لمسارات data/ أو storage/ بس (رفض أي مسار // تاني صراحة) — حتى لو حد حاول يستغل الـWorker نفسه لقراءة شي تاني // بالمستودع. // ============================================================================ const GITHUB_OWNER = 'r3mamo-del'; const GITHUB_REPO = 'raqeeb-data'; export default { async fetch(request, env) { const url = new URL(request.url); const path = decodeURIComponent(url.pathname.replace(/^\//, '')); // طلبات OPTIONS (preflight بمتصفحات) تُجاب فورًا بدون أي معالجة ثانية. if (request.method === 'OPTIONS') { return new Response(null, { headers: corsHeaders({}) }); } // مسارات الإعدادات المشتركة (كلمة مرور المشرف، المستهدفات، إلخ) — // تُحفظ بـ Cloudflare KV، تُقرأ وتُكتب من أي جهاز يفتح التطبيق. if (path.startsWith('storage/')) { return handleStorage(request, env, path.slice('storage/'.length)); } // باقي الطلبات: قراءة بيانات المستشفيات من GitHub — مسارات data/ بس. if (!path.startsWith('data/') || path.includes('..')) { return new Response(JSON.stringify({ error: 'مسار غير مسموح' }), { status: 403, headers: corsHeaders({ 'Content-Type': 'application/json; charset=utf-8' }), }); } const githubUrl = `https://api.github.com/repos/${GITHUB_OWNER}/${GITHUB_REPO}/contents/${encodeURIComponent(path)}?_cb=${Date.now()}`; let res; try { res = await fetch(githubUrl, { headers: { // env.GITHUB_PAT — Secret مُعرَّف بإعدادات Cloudflare نفسها، // غير مرئي بهذا الكود ولا بأي سجلّ إطلاقًا. Authorization: `token ${env.GITHUB_PAT}`, Accept: 'application/vnd.github.v3+json', 'User-Agent': 'tibyan-proxy-worker', }, }); } catch (e) { return new Response(JSON.stringify({ error: 'فشل الاتصال بـGitHub' }), { status: 502, headers: corsHeaders({ 'Content-Type': 'application/json; charset=utf-8' }), }); } if (!res.ok) { return new Response(JSON.stringify({ error: `فشل جلب الملف (${res.status})` }), { status: res.status, headers: corsHeaders({ 'Content-Type': 'application/json; charset=utf-8' }), }); } const meta = await res.json(); // ملفات أكبر من ~1 ميجابايت — GitHub يرجّع download_url بدل content // مباشرة (نفس القيد الأصلي بالتطبيق). if (!meta.content && meta.download_url) { const rawRes = await fetch(`${meta.download_url}&_cb2=${Date.now()}`); if (!rawRes.ok) { return new Response(JSON.stringify({ error: 'فشل تحميل الملف الكبير' }), { status: 502, headers: corsHeaders({ 'Content-Type': 'application/json; charset=utf-8' }), }); } const text = await rawRes.text(); return new Response(text, { headers: corsHeaders({ 'Content-Type': 'application/json; charset=utf-8' }) }); } if (!meta.content) { return new Response(JSON.stringify({ error: 'الملف بلا بيانات' }), { status: 502, headers: corsHeaders({ 'Content-Type': 'application/json; charset=utf-8' }), }); } // فك ترميز Base64 (يدعم النصوص العربية بشكل صحيح). const decoded = new TextDecoder('utf-8').decode( Uint8Array.from(atob(meta.content.replace(/\n/g, '')), c => c.charCodeAt(0)) ); return new Response(decoded, { headers: corsHeaders({ 'Content-Type': 'application/json; charset=utf-8' }) }); }, }; // إعدادات مشتركة (KV) — مفتاح-قيمة بسيط، بدون أي معالجة إضافية. القيمة // دائمًا نص عادي (التطبيق نفسه مسؤول عن أي تحويل JSON لو احتاج). async function handleStorage(request, env, key) { if (!key) { return new Response(JSON.stringify({ error: 'مفتاح مطلوب' }), { status: 400, headers: corsHeaders({ 'Content-Type': 'application/json; charset=utf-8' }), }); } if (request.method === 'GET') { const value = await env.TIBYAN_KV.get(key); if (value === null) { return new Response(JSON.stringify({ error: 'غير موجود' }), { status: 404, headers: corsHeaders({ 'Content-Type': 'application/json; charset=utf-8' }), }); } return new Response(JSON.stringify({ value }), { headers: corsHeaders({ 'Content-Type': 'application/json; charset=utf-8' }), }); } if (request.method === 'POST') { let body; try { body = await request.json(); } catch (e) { return new Response(JSON.stringify({ error: 'محتوى غير صالح' }), { status: 400, headers: corsHeaders({ 'Content-Type': 'application/json; charset=utf-8' }), }); } await env.TIBYAN_KV.put(key, String(body.value ?? '')); return new Response(JSON.stringify({ ok: true }), { headers: corsHeaders({ 'Content-Type': 'application/json; charset=utf-8' }), }); } return new Response(JSON.stringify({ error: 'طريقة غير مدعومة' }), { status: 405, headers: corsHeaders({ 'Content-Type': 'application/json; charset=utf-8' }), }); } function corsHeaders(extra) { return { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Methods': 'GET, POST, OPTIONS', 'Access-Control-Allow-Headers': 'Content-Type', ...extra, }; }
